\section{Theory}
% =============================================================================

\begin{frame}{Key Definitions}
  \begin{definitionbox}[Causal Effect]
    The causal effect of treatment $D$ on outcome $Y$ is:
    \[
      \tau = \E[Y(1) - Y(0)]
    \]
    where $Y(1)$ and $Y(0)$ are potential outcomes.
  \end{definitionbox}
\end{frame}

\begin{frame}{Research Question}
  \begin{questionbox}
    Under what conditions can we identify $\tau$ from observational data?
  \end{questionbox}

  \vspace{1em}

  \begin{answerbox}
    With a valid instrument $Z$ satisfying relevance, independence, and the exclusion restriction \citegray{Angrist1996_iv}.
  \end{answerbox}
\end{frame}

\begin{frame}{Identification Strategy}
  \begin{methodbox}
    \textbf{Two-Stage Least Squares (2SLS)}
    \begin{enumerate}
      \item First stage: regress $D$ on $Z$ and covariates $X$
      \item Second stage: regress $Y$ on $\hat{D}$ and $X$
    \end{enumerate}
  \end{methodbox}

  \vspace{1em}

  \begin{insightbox}
    The IV estimate is consistent under LATE assumptions, even when OLS is biased by endogeneity.
  \end{insightbox}
\end{frame}
